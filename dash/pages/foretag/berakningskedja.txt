def get_current_user_info() -> Dict[str, Any]:
    """
    Hämtar info om inloggad användare från flask_session.
    
    Returns:
        Dict med user_dmu, company_name, org
    """
    user_dmu = flask_session.get('user_dmu')
    org = flask_session.get('org', 'unknown')
    
    # Hämta företagsnamn från reconciliation
    try:
        company_info = load_reconciliation_foretag_info()
        company_name = company_info.get('company_name', 'Ditt företag')
    except:
        company_name = 'Ditt företag'
    
    return {
        'user_dmu': user_dmu,
        'company_name': company_name,
        'org': org
    }


def load_baseline_data_for_company() -> Tuple[pd.DataFrame, Dict[str, Any]]:
    """
    Laddar baseline capbase_a data för inloggat företag.
    
    Returns:
        Tuple med (DataFrame, status_dict)
    """
    user_info = get_current_user_info()
    user_dmu = user_info['user_dmu']
    
    if user_dmu is None:
        return pd.DataFrame(), {'error': 'Ingen DMU hittades'}
    
    try:
        df = load_dmu_capbase_a(user_dmu)
        
        if df.empty:
            return df, {'error': f'Ingen data för DMU {user_dmu}'}
        
        # Validera input
        validation = validate_input_data(df)
        
        status = {
            'rows': len(df),
            'networks': df['id_network'].nunique() if 'id_network' in df.columns else 0,
            'categories': df['cat_encode'].nunique() if 'cat_encode' in df.columns else 0,
            'validation': validation
        }
        
        return df, status
        
    except Exception as e:
        return pd.DataFrame(), {'error': str(e)}


def prepare_lifetime_table_data(df: pd.DataFrame) -> pd.DataFrame:
    """
    Förbereder data för lifetime-editering tabell.
    Aggregerar till kategori-nivå.
    
    Args:
        df: capbase_a DataFrame
        
    Returns:
        DataFrame med kolumner: Kod, Beskrivning, Ekonomisk livslängd, Maximal livslängd
    """
    if df.empty:
        return pd.DataFrame(columns=['Kod', 'Beskrivning', 'Ekonomisk livslängd', 'Maximal livslängd'])
    
    # Aggregera per kategori
    group_cols = ['cat_encode']
    if 'cat' in df.columns:
        group_cols.append('cat')
    
    agg_df = df.groupby(group_cols).agg({
        'ekdep': 'first',
        'maxdep': 'first'
    }).reset_index()
    
    # Formatera för display
    result = pd.DataFrame({
        'Kod': agg_df['cat_encode'],
        'Beskrivning': agg_df['cat'] if 'cat' in agg_df.columns else agg_df['cat_encode'],
        'Ekonomisk livslängd': agg_df['ekdep'].astype(int),
        'Maximal livslängd': agg_df['maxdep'].astype(int)
    })
    
    return result.sort_values('Kod').reset_index(drop=True)


def apply_lifetime_adjustments_to_data(
    df: pd.DataFrame, 
    adjustments: pd.DataFrame
) -> pd.DataFrame:
    """
    Applicerar lifetime-justeringar på capbase_a data.
    
    Args:
        df: Original capbase_a DataFrame
        adjustments: DataFrame från lifetime-tabell
        
    Returns:
        Justerad DataFrame
    """
    df_adjusted = df.copy()
    
    for _, row in adjustments.iterrows():
        kod = row['Kod']
        new_ekdep = row['Ekonomisk livslängd']
        new_maxdep = row['Maximal livslängd']
        
        mask = df_adjusted['cat_encode'] == kod
        df_adjusted.loc[mask, 'ekdep'] = new_ekdep
        df_adjusted.loc[mask, 'maxdep'] = new_maxdep
    
    return df_adjusted

def render_tab_content(active_tab, steps_data):
    """Renderar innehåll för aktiv tab + progress indicator."""
    user_info = get_current_user_info()
    if not user_info.get('user_dmu'):
        return html.Div("Session data saknas"), None
    
    # Progress indicator
    completed = steps_data.get('completed_steps', []) if steps_data else []
    progress = components.create_step_progress_indicator(completed, total_steps=4)
    
    # Tab content
    if active_tab == 'step5':
        return create_step5_content(), progress
    elif active_tab == 'step6':
        return create_step6_content(), progress
    elif active_tab == 'wacc':
        return create_wacc_content(), progress
    elif active_tab == 'step7':
        return create_step7_content(), progress
    elif active_tab == 'step8':
        return create_step8_content(), progress
    else:
        return html.Div("Okänd tab"), progress

@callback(

     Output('tab-step7', 'disabled'),
     Output('tab-step8', 'disabled')],
    Input('steps-store', 'data')
)
def update_tab_states(steps_data):
    """Uppdaterar vilka tabs som är tillgängliga baserat på completed_steps."""
    if not steps_data:
        return True, True, True
    
    completed = set(steps_data.get('completed_steps', []))
    
    step6_disabled = 5 not in completed
    step7_disabled = 5 not in completed
    step8_disabled = not (6 in completed and 7 in completed)
    
    return step6_disabled, step7_disabled, step8_disabled

@callback(
    Output('lifetime-table-container', 'children'),
    Input('lifetime-collapse', 'is_open')
)
def load_lifetime_table(is_open):
    """Laddar lifetime-tabell när sektionen öppnas."""
    if not is_open:
        return no_update
    
    # Ladda baseline data
    df, status = load_baseline_data_for_company()
    
    if df.empty:
        return dbc.Alert(f"Kunde inte ladda data: {status.get('error', 'Okänt fel')}", color="danger")
    
    # Förbered tabell-data
    lifetime_df = prepare_lifetime_table_data(df)
    
    return components.create_lifetime_table(lifetime_df, table_id='lifetime-table')


@callback(
    Output('lifetime-validation-output', 'children'),
    Output('steps-store', 'data', allow_duplicate=True),
    Input('lifetime-table-apply-btn', 'n_clicks'),
    [State('lifetime-table', 'data'),
     State('steps-store', 'data')],
    prevent_initial_call=True
)
def apply_lifetime_adjustments(n_clicks, table_data, steps_data):
    """Applicerar lifetime-justeringar och invaliderar senare steg."""
    if not n_clicks or not table_data:
        raise PreventUpdate
    
    # Konvertera till DataFrame
    adj_df = pd.DataFrame(table_data)
    
    # Validering: maxdep måste vara >= ekdep
    invalid = adj_df[adj_df['Maximal livslängd'] < adj_df['Ekonomisk livslängd']]
    if not invalid.empty:
        return dbc.Alert(
            f"Fel: {len(invalid)} kategori(er) har maximal livslängd < ekonomisk livslängd",
            color="danger"
        ), no_update
    
    # Spara justeringar server-side
    user_info = get_current_user_info()
    state_management.save_lifetime_adjustments(
        user_info['user_dmu'],
        {'adjustments': table_data}
    )
    
    # Invalidera Step 5-8
    user_info = get_current_user_info()
    state_management.clear_steps_from(user_info['user_dmu'], start_step=5)
    steps_data = state_management.invalidate_steps_from(steps_data, start_step=5)
    
    return dbc.Alert("Justeringar applicerade! Kör Steg 5 igen.", color="success"), steps_data


@callback(
    Output('lifetime-table', 'data'),
    Input('lifetime-table-reset-btn', 'n_clicks'),
    prevent_initial_call=True
)
def reset_lifetime_adjustments(n_clicks):
    """Återställer lifetime-tabell till baseline."""
    if not n_clicks:
        raise PreventUpdate
    
    # Rensa sparade justeringar
    user_info = get_current_user_info()
    state_management.clear_lifetime_adjustments(user_info['user_dmu'])
    
    # Ladda baseline igen
    df, _ = load_baseline_data_for_company()
    lifetime_df = prepare_lifetime_table_data(df)
    
    return lifetime_df.to_dict('records')

@app.long_callback(
    [Output('steps-store', 'data', allow_duplicate=True),
     Output('step5-status', 'children'),
     Output('step5-results', 'children')],
    Input('run-step5-btn', 'n_clicks'),
    State('steps-store', 'data'),
    running=[
        (Output('run-step5-btn', 'disabled'), True, False),
        (Output('step5-loading', 'children'), 
         [dbc.Spinner(size="lg"), html.P("Beräknar åldrar och NUAV... Detta kan ta 10-30 sekunder.")],
         None)
    ],
    prevent_initial_call=True
)
def run_step5(n_clicks, steps_data):
    """Kör Steg 5: Åldrar och NUAV beräkning."""
    if not n_clicks:
        raise PreventUpdate
    
    user_info = get_current_user_info()
    user_dmu = user_info['user_dmu']
    
    try:
        # Ladda baseline data
        df, status = load_baseline_data_for_company()
        
        if df.empty:
            return (
                steps_data,
                dbc.Alert(f"Fel: {status.get('error', 'Kunde inte ladda data')}", color="danger"),
                None
            )
        
        # Applicera lifetime-justeringar om de finns
        adjustments = state_management.load_lifetime_adjustments(user_dmu)
        if adjustments:
            adj_df = pd.DataFrame(adjustments['adjustments'])
            df = apply_lifetime_adjustments_to_data(df, adj_df)
        
        # KÖR BERÄKNING (denna tar tid!)
        result_df = calculate_ages_and_nuav(df)
        
        # Spara server-side
        state_management.save_step_dataframe(user_dmu, 5, result_df)
        
        # Uppdatera Store
        steps_data = state_management.update_store_step_completion(steps_data, 5)
        
        # Skapa resultat-visning
        status_alert = dbc.Alert("✅ Steg 5 slutfört!", color="success")
        
        results = html.Div([
            html.H4("Resultat", className="mt-3"),
            dbc.Row([
                dbc.Col([
                    components.create_metric_card(
                        "Komponenter",
                        len(result_df),
                        subtitle="Totalt antal"
                    )
                ], width=3),
                dbc.Col([
                    components.create_metric_card(
                        "Tidsperioder",
                        8,
                        subtitle="H1-H2 2024-2027"
                    )
                ], width=3),
                dbc.Col([
                    components.create_metric_card(
                        "NUAV ordinarie 2024H1",
                        f"{result_df['nuav_ord_229'].sum():,.0f} tkr" if 'nuav_ord_229' in result_df.columns else "N/A"
                    )
                ], width=3)
            ])
        ])
        
        return steps_data, status_alert, results
        
    except Exception as e:
        return (
            steps_data,
            dbc.Alert(f"Fel i Steg 5: {str(e)}", color="danger"),
            None
        )

@callback(
    Output('wacc-input-form', 'children'),
    Output('wacc-results', 'children'),
    [Input('calc-wacc-btn', 'n_clicks'),
     Input('reset-wacc-btn', 'n_clicks')],
    [State('wacc-rf-nom', 'value'),
     State('wacc-mrp', 'value'),
     State('wacc-infl', 'value'),
     State('wacc-credit', 'value'),
     State('wacc-debt-share', 'value'),
     State('wacc-tax-rate', 'value'),
     State('wacc-beta-mode', 'value')],
    prevent_initial_call=False
)
def handle_wacc_calculation(calc_clicks, reset_clicks, rf_nom, mrp, infl, credit, 
                           debt_share, tax_rate, beta_mode):
    """Hanterar WACC-beräkning och reset."""
    
    ctx = dash.callback_context
    triggered_id = ctx.triggered[0]['prop_id'].split('.')[0] if ctx.triggered else None
    
    # Defaults (Ei-standard)
    defaults = {
        'rf_nom': 0.0287,
        'mrp': 0.0668,
        'infl': 0.0202,
        'credit': 0.0114,
        'debt_share': 0.36,
        'tax_rate': 0.206,
        'beta_mode': 'beta_a',
        'beta_a': 0.37,
        'beta_e': 0.54
    }

    if calc_clicks is None and reset_clicks is None:
        form = components.create_wacc_input_form('wacc', defaults)
        return form, None
    
    # Reset till defaults
    if triggered_id == 'reset-wacc-btn':
        form = components.create_wacc_input_form('wacc', defaults)
        return form, None
    
    # Initial load
    if not calc_clicks:
        form = components.create_wacc_input_form('wacc', defaults)
        return form, None
    
    # Beräkna WACC
    try:
        # Bestäm beta
        beta_a = defaults['beta_a'] if beta_mode == 'beta_a' else None
        beta_e = defaults['beta_e'] if beta_mode == 'beta_e' else None
        
        # Skapa inputs
        wacc_inputs = EiWaccInputs(
            rf_nominal=rf_nom,
            mrp_nominal=mrp,
            inflation=infl,
            credit_spread=credit,
            debt_share=debt_share,
            tax_rate=tax_rate,
            beta_asset=beta_a,
            beta_equity=beta_e
        )
        
        # Beräkna
        re_nom, rd_nom, wacc_nom, wacc_real = ei_wacc_real_pre_tax(wacc_inputs)
        
        # Result cards
        result_cards = components.create_wacc_result_cards(re_nom, rd_nom, wacc_nom, wacc_real)
        
        form = components.create_wacc_input_form('wacc', {
            'rf_nom': rf_nom,
            'mrp': mrp,
            'infl': infl,
            'credit': credit,
            'debt_share': debt_share,
            'tax_rate': tax_rate,
            'beta_mode': beta_mode,
            'beta_a': defaults['beta_a'],
            'beta_e': defaults['beta_e']
        })
        
        return form, result_cards
        
    except Exception as e:
        form = components.create_wacc_input_form('wacc', defaults)
        error = dbc.Alert(f"Fel i beräkning: {str(e)}", color="danger")
        return form, error
