# Klistras in i tab2 under policy playground

# === TAB 2: KPI-simulering ===
    with tab2:
        st.markdown("#### Simulera förmögenhetsbevarande värdering (KPI-indexering)")

        # Ladda samplet med anskaffningsvärde
        @st.cache_data
        def load_kpi_sample():
            return pd.read_parquet("kapitalbas_filer/final_capbase_sample.parquet")

        from kapitalbas_app.kpi_simulering import simulera_kapitalbas_kpi

        kpi_df = load_kpi_sample()
        if network_choice != "Alla":
            kpi_df = kpi_df[kpi_df["id_network"] == int(network_choice)]

        result_df = simulera_kapitalbas_kpi(kpi_df)

        # === FELSÖKNING ===
        with st.expander("🛠 Visa felsökning"):
            st.write("📋 Datatyper i kpi_df:")
            st.write(kpi_df.dtypes)

            st.write("🔢 Antal rader före simulering:", len(kpi_df))

            year_map = {
                227.0: 2015, 228.0: 2016, 229.0: 2017,
                230.0: 2018, 231.0: 2019, 232.0: 2020,
                233.0: 2021, 234.0: 2022, 235.0: 2023, 236.0: 2024
            }

            st.write("⛔ time_invest som INTE matchar mapping:")
            st.write(kpi_df[~kpi_df["time_invest"].isin(list(year_map.keys()))]["time_invest"].unique())



            st.write("🧪 Unika time_invest:", kpi_df["time_invest"].dropna().unique())
            st.write("📈 Antal rader med anskaffningsvärde > 0:", (kpi_df["anskaffningsvärde"] > 0).sum())

            st.write("📊 Unika year_invest i resultatet:", result_df["year_invest"].dropna().unique())
            st.write("✅ Antal rader i result_df:", len(result_df))

            for col in result_df.columns:
                if result_df[col].dtype == "object":
                    result_df[col] = result_df[col].astype(str)
            
            st.dataframe(result_df.head(10))
            

        # KPI-mått
        total_nuav = result_df["nuav"].sum()
        total_kpi = result_df["kapitalbas_kpi"].sum()
        total_diff = result_df["diff_kpi_vs_nuav"].sum()

        col1, col2, col3 = st.columns(3)
        col1.metric("Kapitalbas (nuav)", f"{total_nuav/1_000_000:,.1f} MSEK")
        col2.metric("Kapitalbas (KPI)", f"{total_kpi/1_000_000:,.1f} MSEK")
        col3.metric("Skillnad", f"{total_diff/1_000_000:,.1f} MSEK")

        # Altair: jämförelse per kategori
        cat_comp = result_df.groupby("cat")[["nuav", "kapitalbas_kpi"]].sum().reset_index()
        cat_melted = cat_comp.melt(id_vars="cat", var_name="Typ", value_name="Varde")

        chart = alt.Chart(cat_melted).mark_bar().encode(
            x=alt.X("cat:N", title="Kategori"),
            y=alt.Y("Varde:Q", title="MSEK"),
            color="Typ:N"
        ).properties(width=700, height=400)

        st.altair_chart(chart, use_container_width=True)

        # Export-knapp
        st.markdown("### Exportera resultat")
        to_export = result_df.copy()
        buffer = io.BytesIO()
        with pd.ExcelWriter(buffer, engine="xlsxwriter") as writer:
            to_export.to_excel(writer, sheet_name="KPI_simulering", index=False)
        buffer.seek(0)

        st.download_button(
            label="📅 Ladda ner som Excel",
            data=buffer,
            file_name=f"kpi_simulering_nät_{network_choice}.xlsx",
            mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        )
