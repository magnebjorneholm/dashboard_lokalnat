 # === TAB 3: Tail-andelar ===

        # === Ladda sample och totalsats
        sample_df = pd.read_parquet("kapitalbas_filer/capbase_compress_tail_sample.parquet")
        total_df = pd.read_parquet("kapitalbas_filer/capbase_compress_tail.parquet")

        # === Nätkoder och namn (kan bytas ut mot labels)
        nätkoder = sample_df["id_network"].unique().tolist()
        nätkoder.sort()
        nätnamn = {3035: "Stort nät", 160: "Medelstort nät", 7: "Kommunalt nät"}

        # === Funktion: beräkna andelar
        def beräkna_tail_andelar(df, nät_namn="Totalt"):
            nuav_ord_cols = [c for c in df.columns if c.startswith("nuav_ord_")]
            nuav_tail_cols = [c for c in df.columns if c.startswith("nuav_tail_")]
            dep_ord_cols = [c for c in df.columns if c.startswith("dep_ord_")]
            dep_tail_cols = [c for c in df.columns if c.startswith("dep_tail_")]

            nuav_ord_sum = df[nuav_ord_cols].sum().sum()
            nuav_tail_sum = df[nuav_tail_cols].sum().sum()
            dep_ord_sum = df[dep_ord_cols].sum().sum()
            dep_tail_sum = df[dep_tail_cols].sum().sum()

            kapitalbas_andel = nuav_tail_sum / (nuav_ord_sum + nuav_tail_sum) if (nuav_ord_sum + nuav_tail_sum) > 0 else 0
            dep_andel = dep_tail_sum / (dep_ord_sum + dep_tail_sum) if (dep_ord_sum + dep_tail_sum) > 0 else 0

            return {
                "Nät": nät_namn,
                "Tailandel Kapitalbas": kapitalbas_andel * 100,
                "Tailandel Avskrivning": dep_andel * 100,
            }

        # === Samla resultat
        resultat = []

        # Per nät
        for nät in nätkoder:
            namn = nätnamn.get(nät, str(nät))
            df_nät = sample_df[sample_df["id_network"] == nät]
            resultat.append(beräkna_tail_andelar(df_nät, nät_namn=namn))

        # Total
        resultat.append(beräkna_tail_andelar(total_df, nät_namn="Alla nät"))

        valda_nät = st.multiselect("Välj upp till 10 nät att jämföra mot totalen", alla_nät, max_selections=10)

            # === Summera tailandel kapitalbas per nät
            rows = []
            for nät in valda_nät:
                sub = jämförelse_df[jämförelse_df["id_network"] == nät]
                tail_cols = [c for c in sub.columns if c.startswith("nuav_tail_")]
                ord_cols = [c for c in sub.columns if c.startswith("nuav_ord_")]
                tail_sum = sub[tail_cols].sum().sum()
                ord_sum = sub[ord_cols].sum().sum()
                andel = tail_sum / (tail_sum + ord_sum) if (tail_sum + ord_sum) > 0 else 0
                rows.append({"id_network": nät, "Tailandel Kapitalbas": andel * 100})

            jämförelse_resultat = pd.DataFrame(rows)

            # === Hämta totalandel
            tail_cols = [c for c in full_df.columns if c.startswith("nuav_tail_")]
            ord_cols = [c for c in full_df.columns if c.startswith("nuav_ord_")]
            tail_total = full_df[tail_cols].sum().sum()

            total_andel = tail_total / (tail_total + ord_total) if (tail_total + ord_total) > 0 else 0
